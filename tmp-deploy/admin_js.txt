/* ================= 課程報名／排課（官網報名） ================= */
const CS=["待排課","已排課","改期","已完課","沒來","退款","取消"],PS=["未收款","已收訂金","已收全額","已退款"];
const CSC={"待排課":"warn","已排課":"ok","改期":"acc","已完課":"","沒來":"bad","退款":"","取消":"","上完待確認":"bad"};
const PSC={"未收款":"bad","已收訂金":"warn","已收全額":"ok","已退款":""};
let CO=null,coF="all",coSel=null,coEd=null;
function money(n){return "NT$"+Number(n||0).toLocaleString()}
function coDue(r){return r.coachOnly||r.state==="取消"||r.state==="退款"||r.pay==="已退款"?0:Math.max(0,r.total-r.paid)}
function coRank(r){return r.show==="上完待確認"?0:(r.state==="待排課"||r.state==="改期")?1:r.state==="已排課"?2:3}
function coSort(a,b){const x=coRank(a),y=coRank(b);if(x!==y)return x-y;if(x===1)return a.ts<b.ts?-1:1;if(x===2)return a.ds<b.ds?-1:1;return a.ts<b.ts?1:-1}
const COF=[["all","全部"],["待排課","待排課"],["已排課","已排課"],["上完待確認","上完待確認"],["改期","改期"],["已完課","已完課"],["沒來","沒來"],["退款","退款"],["取消","取消"],["$未收款","未收款"],["$已收訂金","已收訂金"],["coach","陪跑班登記"]];
function coMatch(r,f){if(f==="coach")return r.coachOnly;if(r.coachOnly)return false;if(f==="all")return true;if(f[0]==="$")return r.pay===f.slice(1)&&r.state!=="取消"&&r.state!=="退款";if(f==="上完待確認")return r.show===f;return r.state===f}
async function courses(id){CO=await run("adCourses");coSel=id||coSel;coRender()}
function coRender(){const R=CO.rows,L=R.filter(r=>!r.coachOnly),mon=CO.today.slice(0,7);
  const pend=L.filter(r=>r.state==="待排課"||r.state==="改期").length,soon=L.filter(r=>r.state==="已排課"&&r.d>=0&&r.d<=7).length,chk=L.filter(r=>r.show==="上完待確認").length;
  const due=L.reduce((s,r)=>s+coDue(r),0),got=L.filter(r=>r.payDate&&r.payDate.slice(0,7)===mon&&r.pay!=="已退款").reduce((s,r)=>s+r.paid,0);
  const list=R.filter(r=>coMatch(r,coF)).sort(coSort);if(coSel&&!R.find(r=>r.id===coSel))coSel=null;
  V.innerHTML=`<div class="crumb"><a href="#home">所有班級</a>&gt;<span>課程報名／排課</span></div>
  <div class="row sp"><h1>課程報名／排課</h1><span class="muted">官網報名的工作坊、Claude 課</span></div>
  <div class="stats"><div class="stat ${pend?"hot":""}"><div class="n">${pend}</div><div class="muted">待排課</div></div><div class="stat"><div class="n">${soon}</div><div class="muted">7 天內上課</div></div>
   <div class="stat ${chk?"hot":""}"><div class="n">${chk}</div><div class="muted">上完待確認</div></div><div class="stat ${due?"hot":""}"><div class="n">${money(due)}</div><div class="muted">待收款</div></div>
   <div class="stat"><div class="n">${money(got)}</div><div class="muted">本月已收</div></div></div>
  <div class="chips">${COF.map(f=>{const n=R.filter(r=>coMatch(r,f[0])).length;return `<button class="pill ${coF===f[0]?"acc":""}" data-cf="${f[0]}">${f[1]} ${n}</button>`}).join("")}</div>
  <div class="cog"><div class="wrap" style="gap:6px">${list.map(r=>`<button class="coi ${r.id===coSel?"on":""}" data-co="${r.id}">
    <div class="row sp"><span class="nm">${esc(r.name)}${r.nick?`<small>${esc(r.nick)}</small>`:""}</span><span class="row" style="gap:4px"><span class="pill ${CSC[r.show]||""}">${r.show}</span><span class="pill ${PSC[r.pay]||""}">${r.pay}${r.paid&&r.pay!=="已收全額"?" "+r.paid.toLocaleString():""}</span></span></div>
    <small>${esc(r.items)}・${r.ds?fmt(r.ds)+(r.time?" "+esc(r.time):""):"還沒約"}・LINE ${r.line?"已綁":"未綁"}${r.r5||r.r1?"・提醒 "+(r.r5?"前5天✓":"")+(r.r1?" 前1天✓":""):""}・報名 ${md(r.ts)}</small></button>`).join("")||'<div class="card"><div class="muted">這裡沒有資料</div></div>'}</div>
   <div id="coEd">${coSel?"":'<div class="card"><div class="muted">點左邊的學員，在這裡排課、記收款和完課。</div></div>'}</div></div>`;
  document.querySelectorAll("[data-cf]").forEach(b=>b.onclick=()=>{coF=b.dataset.cf;coRender()});
  document.querySelectorAll("[data-co]").forEach(b=>b.onclick=()=>{coSel=b.dataset.co;coRender();if(innerWidth<820)$("coEd").scrollIntoView({behavior:"smooth"})});
  if(coSel)coEditor()}
function coEditor(){const r=CO.rows.find(x=>x.id===coSel);coEd={state:r.state,pay:r.pay};
  const seg=(id,arr,v)=>`<div class="seg" id="${id}">${arr.map(x=>`<button type="button" class="${x===v?"on":""}" data-v="${x}">${x}</button>`).join("")}</div>`;
  const auto=[[r.ts,"官網報名"],r.lineAt?[r.lineAt.replace("T"," "),"LINE 已綁定"]:null,r.r5?[r.r5.slice(0,16).replace("T"," "),"前 5 天提醒已寄（"+r.r5.slice(20)+"）"]:null,r.r1?[r.r1.slice(0,16).replace("T"," "),"前 1 天提醒已寄（"+r.r1.slice(20)+"）"]:null].filter(Boolean).map(x=>x[0].slice(5).replace("-","/")+"　"+x[1]);
  $("coEd").innerHTML=`<div class="card">
   <div><h2>${esc(r.name)}${r.nick?"（"+esc(r.nick)+"）":""}</h2><div class="muted">${esc(r.itemsFull).replace(/\n/g,"<br>")}<br>合計 ${money(r.total)}${r.invoice?"（要發票，工作坊另加 5%）":""}・報名 ${md(r.ts)}</div>
    <div class="muted">${esc(r.email)}${r.phone?"・"+esc(r.phone):""}${r.lineName?"・LINE "+esc(r.lineName):""}・${r.line?"LINE 提醒已綁":"LINE 提醒未綁"}</div>
    ${r.times?`<div class="muted">方便時段：${esc(r.times)}</div>`:""}${r.note?`<div class="muted">問題：${esc(r.note)}</div>`:""}</div>
   <div><h3>課程狀態</h3>${seg("cState",CS,r.state)}</div>
   <div><h3>排課</h3><div class="row"><label>上課日期<input id="cDate" type="date" value="${esc(r.ds)}"></label><label>時間<input id="cTime" value="${esc(r.time)}" placeholder="例：14:00–17:00"></label></div>
    <label>地點（會帶進提醒信）<input id="cPlace" value="${esc(r.place)}" placeholder="例：高雄工作室／線上 Google Meet"></label></div>
   <div><h3>收款</h3>${seg("cPay",PS,r.pay)}<div class="row" style="margin-top:8px"><label>已收金額<input id="cPaid" type="number" min="0" value="${r.paid||""}" placeholder="0"></label><label>方式<input id="cWay" value="${esc(r.payWay)}" placeholder="例：轉帳末五碼 12345"></label></div>
    <div class="row"><label>收款日<input id="cPDate" type="date" value="${esc(r.payDate)}"></label><label>尾款<input id="cDue" disabled></label></div></div>
   <div><h3>完課</h3><div class="row"><label>完課日期<input id="cDone" type="date" value="${esc(r.doneDate)}"></label></div>
    <div class="row"><label class="ck"><input type="checkbox" id="cCert" ${r.cert?"checked":""}>證書已寄</label><label class="ck"><input type="checkbox" id="cSurv" ${r.survey?"checked":""}>已傳課後問卷</label><label class="ck"><input type="checkbox" id="cCoach" ${r.coachInt?"checked":""}>有興趣陪跑班</label></div></div>
   <label>備註<textarea id="cMemo" placeholder="例：改期原因、要帶的素材">${esc(r.memo)}</textarea></label>
   <div class="note">只有「已排課」會寄課前提醒（前 5 天、前 1 天 12:00）。改期會先暫停提醒，填新日期後自動變回已排課。</div>
   <div class="row"><button id="cSave">儲存</button><span class="muted" id="cHint"></span></div>
   <div><h3>紀錄</h3><div class="colog">${esc([r.log].concat(auto.reverse()).filter(Boolean).join("\n"))}</div></div></div>`;
  const upDue=()=>{const p=+$("cPaid").value||0;$("cDue").value=coEd.pay==="已退款"?"—":money(Math.max(0,r.total-p))};upDue();$("cPaid").oninput=upDue;
  const pick=(id,key,fn)=>$(id).querySelectorAll("button").forEach(b=>b.onclick=()=>{coEd[key]=b.dataset.v;$(id).querySelectorAll("button").forEach(x=>x.classList.toggle("on",x===b));fn&&fn(b.dataset.v)});
  pick("cState","state",v=>{if(v==="改期"){$("cDate").value="";$("cHint").textContent="填新日期就會自動變回已排課"}else $("cHint").textContent="";
    if(v==="已完課"&&!$("cDone").value)$("cDone").value=$("cDate").value||CO.today;
    if(v==="退款"){coEd.pay="已退款";$("cPay").querySelectorAll("button").forEach(x=>x.classList.toggle("on",x.dataset.v==="已退款"));upDue()}});
  pick("cPay","pay",v=>{if(v==="已收全額"&&!+$("cPaid").value)$("cPaid").value=r.total;if((v==="已收訂金"||v==="已收全額")&&!$("cPDate").value)$("cPDate").value=CO.today;upDue()});
  $("cSave").onclick=async()=>{busy($("cSave"),1,"儲存中…");
    try{const n=await run("adSaveCourse",r.id,{state:coEd.state,date:$("cDate").value,time:$("cTime").value,place:$("cPlace").value,pay:coEd.pay,paid:$("cPaid").value,payWay:$("cWay").value,payDate:$("cPDate").value,
      doneDate:$("cDone").value,cert:$("cCert").checked,survey:$("cSurv").checked,coachInt:$("cCoach").checked,memo:$("cMemo").value});
      CO.rows[CO.rows.findIndex(x=>x.id===n.id)]=n;toast("已儲存 "+n.name+"："+n.show);coRender()}catch(e){fail(e);busy($("cSave"),0)}}}
